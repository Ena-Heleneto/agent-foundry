import moment from 'moment'

/**
 * UTC 发布时间戳使用固定的毫秒精度，T 和 Z 为字面字符。
 */
const releaseTimestampFormat = 'YYYYMMDD[T]HHmmssSSS[Z]'

/**
 * 生成当前 UTC 发布时间戳，或严格校验调用方冻结的时间戳。
 * @param timestamp 已提供的冻结时间戳。缺省时取当前时间。
 * @returns 符合发布格式的 UTC 毫秒时间戳。
 */
export function resolveReleaseTimestamp(timestamp?: string): string {
  if (timestamp === undefined)
    return moment.utc().locale('en').format(releaseTimestampFormat)

  /**
   * 用固定区域设置解析 UTC 日期，回读时校验历日与格式均一致。
   */
  const parsed = moment.utc(timestamp, releaseTimestampFormat, 'en', true)
  if (!parsed.isValid() || parsed.format(releaseTimestampFormat) !== timestamp)
    throw new Error('FOUNDRY_SKILLS_RELEASE_TIMESTAMP 必须是 YYYYMMDDTHHmmssSSSZ 格式的有效 UTC 时间')
  return timestamp
}
