def record_tool(scope, data):
    """跳过提交步骤，从宿主和本次回调保存可取得的工具详情。"""
    arguments = data.get("tool_input")
    if is_commit(arguments):
        return

    with session_lock(scope):
        contents = collect(scope)
        call = data.get("tool_use_id")
        name = data.get("tool_name")
        is_code = is_code_exec(name) or code_wait_cell(name, arguments) is not None
        if call in contents.withheld_calls or (is_code and call not in contents.tool_outputs):
            return
        if contents.commit_result and (not call or commit_verification(arguments)):
            return
        turn = load_turn(scope, contents.inputs)
        if turn is None:
            return

        events = list(contents.events)
        if call:
            # 回调可能早于宿主落盘。仅补齐尚未从宿主读到的调用和结果。
            location = f"工具事件 {call}"
            if call not in contents.tool_privacy:
                name = redacted(str(data.get("tool_name") or "工具"))
                value = redacted(f"工作目录：{data.get('cwd')}\n{pretty(arguments)}")
                events.append(Event(location, f"工具调用 `{name}` · `{call}`", value, identity=f"call:{call}"))
            if call not in contents.tool_outputs:
                omit = contents.tool_privacy.get(call)
                if omit is None:
                    omit = private_output(arguments)
                events.append(captured_output(location, call, data.get("tool_response"), omit, provisional=True))

        failure = link_details(scope, turn, events)
        turn.save(scope, agent_tool_message(scope, data))
        if failure is not None:
            raise failure

def record_prompt(scope, data):
    """在业务开始前保存并读回用户发起的消息或同轮补充。"""
    prompt = data.get("prompt")
    if not isinstance(prompt, str):
        return
    prompt = redacted(prompt)

    with session_lock(scope):
        turn = open_turn(scope)
        if not turn.messages:
            turn.append("发起", prompt)
            document = turn.save(scope)
            recover_subagent_messages(scope, turn.path, document)
            return

        count = sum(label in {"发起", "补充"} for label, _ in turn.messages)
        repeated = turn.messages[-1] in (("发起", prompt), ("补充", prompt))
        # 用宿主输入条数区分重复回调和真实的同文补充。
        if not repeated or len(collect(scope).inputs) >= count:
            turn.append("补充", prompt)
            turn.save(scope)

        """在父轮空闲时保存，活跃父轮交由父会话回调补齐。"""
        deadline = None
        while True:
            with session_lock(scope):
                scope = verify_scope(scope.project, data, scope.cwd, allow_idle=True)
                if scope is None or not parent_directory_matches(scope):
                    return
                if not scope.active:
                    # 下一次主代理业务动作仍须通过同一把锁保存用户输入。
                    self._save_idle_reply(scope, data)
                    return

                    return
            # Stop 已保存父回复，宿主可能尚未完成索引更新。复查期间释放写入锁。
            if deadline is None:
                deadline = monotonic() + 0.5
            remaining = deadline - monotonic()

                return
